**技术实现：**
```javascript
// 替换照片功能
function replacePhoto(index) {
    if (isUploading) {
        showMessage('正在上传中，请稍候...', 'warning');
        return;
    }
    
    if (!navigator.onLine) {
        showMessage('网络连接异常，请检查网络后重试', 'error');
        return;
    }
    
    if (confirm('确定要替换这张照片吗？')) {
        currentPhotoIndex = index;
        uploadPhoto();
    }
}
```

   {
       if (imageUrl.StartsWith("/uploads/"))
