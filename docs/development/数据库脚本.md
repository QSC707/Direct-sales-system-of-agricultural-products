-- 创建数据库
IF NOT EXISTS (SELECT name FROM sys.databases WHERE name = N'FarmDirectSales')
BEGIN
    CREATE DATABASE FarmDirectSales;
END
GO

USE FarmDirectSales;
GO

-- 用户表
CREATE TABLE Users (
    UserId INT IDENTITY(1,1) PRIMARY KEY,
    Username NVARCHAR(50) NOT NULL,
    Password NVARCHAR(100) NOT NULL,
    Salt NVARCHAR(100) NOT NULL,
    Email NVARCHAR(100) NOT NULL,
    Phone NVARCHAR(20),
    Address NVARCHAR(200),
    Role NVARCHAR(20) NOT NULL DEFAULT 'customer',
    RegisterTime DATETIME NOT NULL DEFAULT GETDATE(),
    LastLoginTime DATETIME,
    Status INT NOT NULL DEFAULT 1
);

-- 产品表
CREATE TABLE Products (
    ProductId INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(100) NOT NULL,
    Description NVARCHAR(500),
    Category NVARCHAR(50) NOT NULL,
    Price DECIMAL(10, 2) NOT NULL,
    Stock INT NOT NULL DEFAULT 0,
    ImageUrl NVARCHAR(200),
    CreatedTime DATETIME NOT NULL DEFAULT GETDATE(),
    UpdatedTime DATETIME NOT NULL DEFAULT GETDATE(),
    Status INT NOT NULL DEFAULT 1
);

-- 订单表
CREATE TABLE Orders (
    OrderId INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT NOT NULL,
    OrderNumber NVARCHAR(50) NOT NULL,
    TotalAmount DECIMAL(10, 2) NOT NULL,
    Status NVARCHAR(20) NOT NULL DEFAULT 'pending',
    ShippingAddress NVARCHAR(200) NOT NULL,
    ShippingMethod NVARCHAR(50),
    ShippingFee DECIMAL(10, 2) DEFAULT 0,
    PaymentMethod NVARCHAR(50),
    PaymentStatus NVARCHAR(20) DEFAULT 'unpaid',
    CreatedTime DATETIME NOT NULL DEFAULT GETDATE(),
    PaidTime DATETIME,
    ShippedTime DATETIME,
    CompletedTime DATETIME,
    CancelledTime DATETIME,
    Remark NVARCHAR(500),
    FOREIGN KEY (UserId) REFERENCES Users(UserId)
);

-- 订单明细表
CREATE TABLE OrderItems (
    OrderItemId INT IDENTITY(1,1) PRIMARY KEY,
    OrderId INT NOT NULL,
    ProductId INT NOT NULL,
    Quantity INT NOT NULL,
    Price DECIMAL(10, 2) NOT NULL,
    Subtotal DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (OrderId) REFERENCES Orders(OrderId),
    FOREIGN KEY (ProductId) REFERENCES Products(ProductId)
);

-- 配送区域表
CREATE TABLE DeliveryAreas (
    AreaId INT IDENTITY(1,1) PRIMARY KEY,
    Province NVARCHAR(50) NOT NULL,
    City NVARCHAR(50) NOT NULL,
    District NVARCHAR(50) NOT NULL,
    IsAvailable BIT NOT NULL DEFAULT 1,
    MinOrderAmount DECIMAL(10, 2) DEFAULT 0,
    CreatedTime DATETIME NOT NULL DEFAULT GETDATE(),
    UpdatedTime DATETIME NOT NULL DEFAULT GETDATE()
);

-- 运费规则表
CREATE TABLE ShippingFees (
    FeeId INT IDENTITY(1,1) PRIMARY KEY,
    AreaId INT,
    MinWeight DECIMAL(10, 2) NOT NULL,
    MaxWeight DECIMAL(10, 2) NOT NULL,
    Fee DECIMAL(10, 2) NOT NULL,
    CreatedTime DATETIME NOT NULL DEFAULT GETDATE(),
    UpdatedTime DATETIME NOT NULL DEFAULT GETDATE(),
    FOREIGN KEY (AreaId) REFERENCES DeliveryAreas(AreaId)
);

-- 系统日志表
CREATE TABLE Logs (
    LogId INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT,
    Username NVARCHAR(50),
    IpAddress NVARCHAR(50),
    ActionType NVARCHAR(50) NOT NULL,
    Description NVARCHAR(500),
    ActionTime DATETIME NOT NULL DEFAULT GETDATE(),
    TargetId INT,
    TargetType NVARCHAR(50),
    IsSuccess BIT NOT NULL DEFAULT 1,
    FOREIGN KEY (UserId) REFERENCES Users(UserId)
);
